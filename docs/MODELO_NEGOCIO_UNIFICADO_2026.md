# core/signals.py - Automação de Confiança

from django.db.models.signals import post_save
from django.dispatch import receiver
from django.utils import timezone

# CERTIFICAÇÕES → Trust +20
@receiver(post_save, sender=AssessmentResult)
def on_assessment_passed(sender, instance, created, **kwargs):
    if created and instance.passed:
        user = instance.enrollment.user
        user.cert_trust = min(user.cert_trust + 20, 20)
        user.trust_score = sum([
            user.cert_trust,
            user.market_trust,
            user.kixikila_trust,
            user.reality_trust,
            user.community_trust,
        ])
        user.save()
        
        # Log event para análise
        TrustEvent.objects.create(
            user=user,
            event_type='certification_passed',
            points=20,
            description=f"Certificação INEFOB: {instance.enrollment.program.title}"
        )

# MARKETPLACE → Trust +5 por ordem, +25 por 5 ordens
@receiver(post_save, sender=ServiceOrder)
def on_order_completed(sender, instance, created, **kwargs):
    if instance.status == 'completed':
        # Provider gains trust
        provider = instance.listing.provider
        provider.market_trust = min(provider.market_trust + 5, 25)
        
        # Check milestone: 5 completed orders
        completed_count = provider.service_provider.listings.aggregate(
            total=Count('orders', filter=Q(orders__status='completed'))
        )['total']
        
        if completed_count % 5 == 0:
            provider.trust_score += 20  # Bonus!
        else:
            provider.trust_score += 5
        
        provider.total_earnings += instance.total_price * 0.85  # Platform takes 15%
        provider.save()
        
        TrustEvent.objects.create(
            user=provider,
            event_type='marketplace_order',
            points=5,
            description=f"Ordem completa: {instance.listing.title}"
        )

# KIXIKILA → Trust +20 por payout completo
@receiver(post_save, sender=KixikilaPayout)
def on_kixikila_payout(sender, instance, created, **kwargs):
    if created:
        member = instance.membership.member
        member.kixikila_trust = min(member.kixikila_trust + 20, 20)
        member.total_kixikila_saved += instance.net_amount
        member.trust_score = sum([...])  # Recalc
        member.save()
        
        TrustEvent.objects.create(
            user=member,
            event_type='kixikila_cycle',
            points=20,
            description=f"Ciclo completo Kixikila: {instance.group.name}"
        )

# REALITY → Trust +50 ao vencer
@receiver(post_save, sender=SeasonParticipant)
def on_season_winner(sender, instance, created, **kwargs):
    if instance.status == 'winner':
        user = instance.user
        user.reality_trust = min(user.reality_trust + 50, 25)  # max 25
        user.trust_score = sum([...])
        user.save()
        
        TrustEvent.objects.create(
            user=user,
            event_type='reality_winner',
            points=50,
            description=f"Vencedor - Temporada {instance.season.season_number}"
        )

# AUTO: Recalc trust_level based on trust_score
@receiver(post_save, sender=User)
def update_trust_level(sender, instance, **kwargs):
    if instance.trust_score <= 20:
        instance.trust_level = 'novice'
    elif instance.trust_score <= 50:
        instance.trust_level = 'contributor'
    elif instance.trust_score <= 80:
        instance.trust_level = 'trusted'
    else:
        instance.trust_level = 'leader'
    instance.save(update_fields=['trust_level'])

## 🔌 APIs Unificadas (Implementadas)

Para suportar a harmonia entre módulos, foi criada uma camada "core" com sinais e APIs de consolidação.

- GET `/api/v2/core/me/dashboard/` (auth)
    - Retorna visão integrada do utilizador: trust (score + breakdown), certificações, marketplace, kixikila e temporadas.
- GET `/api/v2/core/me/revenue/` (auth admin/opcional)
    - Resumo de receitas por fonte (commission marketplace, taxa kixikila, etc.).

Integrações automáticas via sinais (auditáveis em `TrustEvent`):
- Certificações: `AssessmentResult(PASS)` → `event_type=cert_pass`, `points=20`.
- Marketplace: `ServiceOrder(status=completed)` → `market_order` (+ receita de comissão 15%).
- Kixikila: `KixikilaPayout(status=completed)` → `kix_cycle` (+ receita `platform_fee`).
- Jogos: `SimulatorSession(finished_at!=null)` → `game_session` (+2 pontos).
- Realities: `Participant(status=winner)` → `reality_winner` (+50 pontos).

Passos para ativação:
1) Migrar a base de dados (novos modelos `core`):
```powershell
Push-Location "C:\apps\Acredita";
& .venv\Scripts\python.exe manage.py makemigrations core;
& .venv\Scripts\python.exe manage.py migrate;
Pop-Location
```
2) Iniciar servidor Django e validar endpoints:
```powershell
Push-Location "C:\apps\Acredita";
& .venv\Scripts\python.exe manage.py runserver
```
3) Verificar no frontend (próxima fase) o consumo do endpoint `/api/v2/core/me/dashboard/` para Homepage/Dashboard unificado.
